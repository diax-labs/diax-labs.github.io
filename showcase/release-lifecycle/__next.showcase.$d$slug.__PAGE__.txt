function scrollOnArrow(event: KeyboardEvent<HTMLElement>) {
  if (event.target !== event.currentTarget) return;
  const direction = event.key === "ArrowRight" ? 1 : event.key === "ArrowLeft" ? -1 : 0;
  if (direction === 0) return;
  event.preventDefault();
  event.currentTarget.scrollBy({ left: direction * Math.max(120, event.currentTarget.clientWidth * 0.7) });
}
