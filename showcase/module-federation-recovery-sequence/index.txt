function MessageRow({ message, index }: { message: RecoveryMessage; index: number }) {
  return (
    <div
      data-sequence-message={message.id}
      data-sequence-kind={message.kind}
      className="grid min-h-[72px] grid-cols-[4rem_repeat(4,minmax(0,1fr))] border-t border-[#d1cec3] first:border-t-0"

      </div>
      {participants.map((participant) => {
        const endpoint = participant.id === message.from || participant.id === message.to;
        return (
          <div key={participant.id} className="relative border-l border-[#c5c2b7]">
            {endpoint ? (
              <span
                {...connectionTarget(endpointId(message.id, participant.id))}

                className="absolute top-1/2 left-1/2 z-10 size-2 -translate-x-1/2 -translate-y-1/2 border border-[#f5f2e9] outline outline-1 outline-[#687476]"
                style={{ backgroundColor: relationColor[message.kind] }}
              />
            ) : null}
          </div>
        );
      })}
    </div>
  );
}

function LegendItem({

  label: string;
  color: string;

        style={{ borderColor: color }}

    </li>
  );
}

function participantIndex(id: ParticipantId): number {
  return participants.findIndex((participant) => participant.id === id);
}

function participantLabel(id: ParticipantId): string {
  return participants.find((participant) => participant.id === id)?.label ?? id;
}

function endpointId(messageId: string, participantId: ParticipantId): string {
  return `${messageId}-${participantId}`;
