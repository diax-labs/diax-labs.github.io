# API Reference (/docs/svelte/props)



Component Props [#component-props]

`FlowDiagram` and `Mindmap` accept the following presentation props. Custom generic payloads remain
available through the lower-level store interface below.

<TypeTable
  type="{
  nodes: { description: 'String-content node definitions', type: 'DiaxNode<string>[]', default: 'required' },
  relations: { description: 'Relations', type: 'DiaxRelation<string>[]', default: 'required' },
  groups: { description: 'Nested semantic boundaries', type: 'DiaxGroup<string>[]', default: '[]' },
  namespace: { description: 'Resolved DOM ID namespace', type: 'string', default: '—' },
  direction: { description: 'Layout direction', type: &#x22;'horizontal' | 'vertical' | 'horizontal-reverse' | 'vertical-reverse'&#x22;, default: &#x22;'horizontal'&#x22; },
  density: { description: 'Spacing', type: &#x22;'compact' | 'comfortable' | 'spacious'&#x22;, default: &#x22;'comfortable'&#x22; },
  centered: { description: 'Center nodes', type: 'boolean', default: 'false' },
  highlightOnHover: { description: 'Highlight connected paths', type: 'boolean', default: 'false' },
  theme: { description: 'Theme preset or object', type: 'ThemeInput' },
  pathType: { description: 'Edge path type', type: &#x22;'bezier' | 'straight' | 'orthogonal'&#x22;, default: &#x22;'bezier'&#x22; },
  strokeColor: { description: 'Edge color', type: 'string' },
  strokeWidth: { description: 'Edge width', type: 'number', default: '1' },
  arrowSize: { description: 'Arrow marker size', type: 'number', default: '6' },
}"
/>
